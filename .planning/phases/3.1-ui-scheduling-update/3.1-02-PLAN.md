    ```javascript
    function formatCronExprs(exprs){
      if(!exprs||exprs.length===0)return'<span class="badge badge-never">Not scheduled</span>';
      var parts=[];
      for(var i=0;i<exprs.length;i++){
        var e=exprs[i].trim();
        // Handle @every interval expressions
        if(e.startsWith('@every ')){
          var interval=e.substring(7).trim();
          parts.push('<span title="'+escapeHtml(e)+'" style="color:#90caf9">Every '+interval+'</span>');
          continue;
        }
        // Handle @daily, @hourly, @weekly shortcuts
        if(e==='@daily'||e==='@midnight'){parts.push('<span title="'+escapeHtml(e)+'" style="color:#90caf9">Daily at midnight</span>');continue}
        if(e==='@hourly'){parts.push('<span title="'+escapeHtml(e)+'" style="color:#90caf9">Every hour</span>');continue}
        if(e==='@weekly'){parts.push('<span title="'+escapeHtml(e)+'" style="color:#90caf9">Weekly on Sunday</span>');continue}
        if(e==='@monthly'){parts.push('<span title="'+escapeHtml(e)+'" style="color:#90caf9">Monthly on the 1st</span>');continue}
        if(e==='@yearly'||e==='@annually'){parts.push('<span title="'+escapeHtml(e)+'" style="color:#90caf9">Yearly on Jan 1</span>');continue}
        // Standard 5-field cron expression
        var p=e.split(/\s+/);
        if(p.length!==5){parts.push(escapeHtml(e));continue}
        var m=p[0],h=p[1],dom=p[2],mon=p[3],dow=p[4],desc='';
        if(m==='*'&&h==='*'&&dom==='*'&&mon==='*'&&dow==='*')desc='Every minute';
        else if(m==='0'&&h==='*'&&dom==='*'&&mon==='*'&&dow==='*')desc='Every hour';
        else if(m==='0'&&dom==='*'&&mon==='*'&&dow==='*'){var hn=parseInt(h);desc=hn===0?'Midnight':hn===12?'Noon':hn>12?'Daily at '+(hn-12)+':00 PM':'Daily at '+hn+':00 AM'}
        else if(m.startsWith('*/'))desc='Every '+m.substring(2)+' minute(s)';
        else if(h.startsWith('*/'))desc='Every '+h.substring(2)+' hours';
        else if(h.includes(',')){desc='At '+h.split(',').map(function(x){var n=parseInt(x);return n===0?'midnight':n===12?'noon':n>12?(n-12)+' PM':n+' AM'}).join(', ');if(m!=='0')desc+=' minute '+m}
        else{var hn2=parseInt(h);desc=(hn2===0?'Midnight':hn2===12?'Noon':hn2>12?'Daily at '+(hn2-12)+':00 PM':'Daily at '+hn2+':00 AM')+(m!=='0'?' at minute '+m:'')}
        parts.push('<span title="'+escapeHtml(e)+'" style="color:#a8b2d1">'+escapeHtml(desc)+'</span>');
      }
      return parts.join('<br>');
    }
    ```

    function rescheduleJob(jobId,currentExprs){
      var promptMsg='Enter cron expression(s) for '+jobId+':\nSeparate multiple expressions with commas.\nExamples:\n  0 2 * * * (daily at 2am)\n  0 */6 * * * (every 6 hours)\n  @every 6h (every 6 hours)';
      var current=currentExprs&&currentExprs.length>0?currentExprs.join(', '):'0 2 * * *';
      var input=prompt(promptMsg,current);
      if(!input||!input.trim())return;
      var exprs=input.split(',').map(function(s){return s.trim()}).filter(function(s){return s.length>0});
      if(exprs.length===0)return;
      fetch('/api/v1/jobs/'+jobId,{method:'PATCH',headers:{'Content-Type':'application/json'},body:JSON.stringify({cronExprs:exprs})}).then(function(r){
        if(r.ok){showToast('Schedule updated for '+jobId,'success');loadDashboard()}else{r.json().then(function(d){showToast('Failed: '+(d.error||r.statusText),'error')})}

    }
